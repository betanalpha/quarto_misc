\tikzmath{
  function xparabola(\y, \xv, \yv, \S) {
    return 0.5 * ( (\xv + \S) - (\y - \yv) * (\y - \yv) / (\S - \xv) );
  };
  function pairint(\xa, \ya, \xb, \yb, \S) {
    \a = \xb - \xa;
    \b = \yb * (\S - \xa) - \ya * (\S - \xb);
    \c = \yb * \yb * (\S - \xa) - \ya * \ya * (\S - \xb) - (\S - \xa) * (\S - \xb) * (\xb - \xa);
    \d = sqrt( \b * \b - \a * \c);
    \yint = (\b + \d) / \a;
    if \yint < \ya then {
      \yint = (\b - \d) / \a;
    };
    if \yint > \yb then {
      \yint = (\b - \d) / \a;
    };
    return \yint;
  };
  function circumcenterx(\xa, \ya, \xb, \yb, \xc, \yc) {
    \D = 2 * ( \xa * (\yb - \yc) + \xb * (\yc - \ya) + \xc * (\ya - \yb) );
    \xcnt = (  (\xa * \xa + \ya * \ya) * (\yb - \yc)
             + (\xb * \xb + \yb * \yb) * (\yc - \ya)
             + (\xc * \xc + \yc * \yc) * (\ya - \yb) ) / \D;
    return \xcnt;
  };
  function circumcentery(\xa, \ya, \xb, \yb, \xc, \yc) {
    \D = 2 * ( \xa * (\yb - \yc) + \xb * (\yc - \ya) + \xc * (\ya - \yb) );
    \ycnt = (  (\xa * \xa + \ya * \ya) * (\xc - \xb)
             + (\xb * \xb + \yb * \yb) * (\xa - \xc)
             + (\xc * \xc + \yc * \yc) * (\xb - \xa) ) / \D;
    return \ycnt;
  };
  function circumradius(\xa, \ya, \xb, \yb, \xc, \yc) {
    \D = 2 * ( \xa * (\yb - \yc) + \xb * (\yc - \ya) + \xc * (\ya - \yb) );
    \xcnt = (  (\xa * \xa + \ya * \ya) * (\yb - \yc)
             + (\xb * \xb + \yb * \yb) * (\yc - \ya)
             + (\xc * \xc + \yc * \yc) * (\ya - \yb) ) / \D;
    \ycnt = (  (\xa * \xa + \ya * \ya) * (\xc - \xb)
             + (\xb * \xb + \yb * \yb) * (\xa - \xc)
             + (\xc * \xc + \yc * \yc) * (\xb - \xa) ) / \D;
    \r = sqrt( (\xcnt - \xa) * (\xcnt - \xa) + (\ycnt - \ya) * (\ycnt - \ya) );
    return \r;
  };
}

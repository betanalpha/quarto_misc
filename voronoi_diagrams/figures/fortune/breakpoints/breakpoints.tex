\tikzmath{
  function xparabola(\y, \xv, \yv, \S) {
    return 0.5 * ( (\xv + \S) - (\y - \yv) * (\y - \yv) / (\S - \xv) );
  };
  function pairint(\xa, \ya, \xb, \yb, \S) {
    \a = \xb - \xa;
    \b = \yb * (\S - \xa) - \ya * (\S - \xb);
    \c = \yb * \yb * (\S - \xa) - \ya * \ya * (\S - \xb) - (\S - \xa) * (\S - \xb) * (\xb - \xa);
    \d = sqrt( \b * \b - \a * \c);
    \yint = (\b + \d) / \a;
    if \yint < \ya then {
      \yint = (\b - \d) / \a;
    };
    if \yint > \yb then {
      \yint = (\b - \d) / \a;
    };
    return \yint;
  };
}
